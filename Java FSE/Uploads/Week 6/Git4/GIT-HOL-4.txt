# ---------------------------------------------------------
# Step 1: Verify master is clean
# ---------------------------------------------------------
git status

# ---------------------------------------------------------
# Step 2: Create new branch
# ---------------------------------------------------------
git checkout -b GitWork

# ---------------------------------------------------------
# Step 3: Create hello.xml
# ---------------------------------------------------------
notepad hello.xml

# ---------------------------------------------------------
# Step 4: Add file
# ---------------------------------------------------------
git add hello.xml

# ---------------------------------------------------------
# Step 5: Commit changes
# ---------------------------------------------------------
git commit -m "Added hello.xml in branch"

# ---------------------------------------------------------
# Step 6: Switch to master
# ---------------------------------------------------------
git checkout master

# ---------------------------------------------------------
# Step 7: Modify hello.xml with different content
# ---------------------------------------------------------
notepad hello.xml

# ---------------------------------------------------------
# Step 8: Commit changes
# ---------------------------------------------------------
git add hello.xml
git commit -m "Updated hello.xml in master"

# ---------------------------------------------------------
# Step 9: Display commit history
# ---------------------------------------------------------
git log --oneline --graph --decorate --all

# ---------------------------------------------------------
# Step 10: Compare differences
# ---------------------------------------------------------
git diff GitWork

# ---------------------------------------------------------
# Step 11: Merge branch
# ---------------------------------------------------------
git merge GitWork

# ---------------------------------------------------------
# Step 12: Resolve merge conflict manually
# ---------------------------------------------------------
notepad hello.xml

# ---------------------------------------------------------
# Step 13: Stage resolved file
# ---------------------------------------------------------
git add hello.xml

# ---------------------------------------------------------
# Step 14: Commit resolved conflict
# ---------------------------------------------------------
git commit -m "Resolved merge conflict"

# ---------------------------------------------------------
# Step 15: Ignore backup files
# ---------------------------------------------------------
notepad .gitignore

# Add:
# *.orig

# ---------------------------------------------------------
# Step 16: Commit updated .gitignore
# ---------------------------------------------------------
git add .gitignore
git commit -m "Ignored backup files"

# ---------------------------------------------------------
# Step 17: List all branches
# ---------------------------------------------------------
git branch

# ---------------------------------------------------------
# Step 18: Delete merged branch
# ---------------------------------------------------------
git branch -d GitWork

# ---------------------------------------------------------
# Step 19: View commit history
# ---------------------------------------------------------
git log --oneline --graph --decorate
