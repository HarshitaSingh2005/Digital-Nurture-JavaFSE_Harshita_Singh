# ---------------------------------------------------------
# Step 1: Verify Git installation
# ---------------------------------------------------------
git --version

# ---------------------------------------------------------
# Step 2: Configure Git username
# ---------------------------------------------------------
git config --global user.name "Prashasti Satpathi"

# ---------------------------------------------------------
# Step 3: Configure Git email
# ---------------------------------------------------------
git config --global user.email "prashastiaries@gmail.com"

# ---------------------------------------------------------
# Step 4: Verify Git configuration
# ---------------------------------------------------------
git config --global --list

# ---------------------------------------------------------
# Step 5: Create a new Git repository
# ---------------------------------------------------------
mkdir GitDemo
cd GitDemo
git init

# ---------------------------------------------------------
# Step 6: Verify repository initialization
# ---------------------------------------------------------
ls -al

# ---------------------------------------------------------
# Step 7: Create a text file
# ---------------------------------------------------------
echo "Welcome to the version control" > welcome.txt

# ---------------------------------------------------------
# Step 8: Verify the file
# ---------------------------------------------------------
ls -al

# ---------------------------------------------------------
# Step 9: View file contents
# ---------------------------------------------------------
cat welcome.txt

# ---------------------------------------------------------
# Step 10: Check repository status
# ---------------------------------------------------------
git status

# ---------------------------------------------------------
# Step 11: Add file to staging area
# ---------------------------------------------------------
git add welcome.txt

# ---------------------------------------------------------
# Step 12: Commit changes
# (If default editor opens, type commit message, save and exit)
# ---------------------------------------------------------
git commit

# OR

git commit -m "Initial Commit"

# ---------------------------------------------------------
# Step 13: Verify working tree is clean
# ---------------------------------------------------------
git status

# ---------------------------------------------------------
# Step 14: Connect remote repository
# ---------------------------------------------------------
git remote add origin https://github.com/RoshmikAgrawal/GitDemo.git

# ---------------------------------------------------------
# Step 15: Pull remote repository
# ---------------------------------------------------------
git pull origin master

# ---------------------------------------------------------
# Step 16: Push local repository to remote repository
# ---------------------------------------------------------
git push origin master
