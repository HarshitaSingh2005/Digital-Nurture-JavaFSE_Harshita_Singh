# ---------------------------------------------------------
# Step 1: Create a new branch
# ---------------------------------------------------------
git branch GitNewBranch

# ---------------------------------------------------------
# Step 2: View all branches
# ---------------------------------------------------------
git branch -a

# ---------------------------------------------------------
# Step 3: Switch to the new branch
# ---------------------------------------------------------
git checkout GitNewBranch

# ---------------------------------------------------------
# Step 4: Create a file
# ---------------------------------------------------------
echo "Branch Content" > branch.txt

# ---------------------------------------------------------
# Step 5: Stage file
# ---------------------------------------------------------
git add branch.txt

# ---------------------------------------------------------
# Step 6: Commit changes
# ---------------------------------------------------------
git commit -m "Added branch file"

# ---------------------------------------------------------
# Step 7: Verify repository status
# ---------------------------------------------------------
git status

# ---------------------------------------------------------
# Step 8: Switch back to master branch
# ---------------------------------------------------------
git checkout master

# ---------------------------------------------------------
# Step 9: Display differences between master and branch
# ---------------------------------------------------------
git diff master GitNewBranch

# ---------------------------------------------------------
# Step 10: Merge branch into master
# ---------------------------------------------------------
git merge GitNewBranch

# ---------------------------------------------------------
# Step 11: Display commit history
# ---------------------------------------------------------
git log --oneline --graph --decorate

# ---------------------------------------------------------
# Step 12: Delete merged branch
# ---------------------------------------------------------
git branch -d GitNewBranch

# ---------------------------------------------------------
# Step 13: Verify repository status
# ---------------------------------------------------------
git status
