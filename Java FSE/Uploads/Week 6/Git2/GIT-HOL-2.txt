# ---------------------------------------------------------
# Step 1: Navigate to repository
# ---------------------------------------------------------
cd GitDemo

# ---------------------------------------------------------
# Step 2: Create a log file
# ---------------------------------------------------------
touch sample.log

# ---------------------------------------------------------
# Step 3: Create a logs directory
# ---------------------------------------------------------
mkdir logs

# ---------------------------------------------------------
# Step 4: Create a file inside logs folder
# ---------------------------------------------------------
touch logs/test.txt

# ---------------------------------------------------------
# Step 5: Create .gitignore file
# ---------------------------------------------------------
touch .gitignore

# ---------------------------------------------------------
# Step 6: Open .gitignore using Notepad
# ---------------------------------------------------------
notepad .gitignore

# ---------------------------------------------------------
# Step 7: Add the following lines inside .gitignore
# Ignore all .log files
# Ignore logs directory
# ---------------------------------------------------------

*.log
logs/

# ---------------------------------------------------------
# Step 8: Save .gitignore and add to staging area
# ---------------------------------------------------------
git add .gitignore

# ---------------------------------------------------------
# Step 9: Commit changes
# ---------------------------------------------------------
git commit -m "Added gitignore"

# ---------------------------------------------------------
# Step 10: Verify ignored files
# ---------------------------------------------------------
git status
